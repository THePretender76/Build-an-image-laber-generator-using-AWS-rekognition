    <script>
        async function uploadImage() {
            const file = document.getElementById('imageInput').files[0];
            const status = document.getElementById('status');
            const LAMBDA_URL = 'YOUR_FUNCTION_URL_HERE';

            if (!file) return alert('Select a file!');
            status.innerText = "Getting permission...";

            // 1. Get the Presigned URL
            const response = await fetch(`${https://sfz2axozliafccsk7jkbtgvbhu0lgsdq.lambda-url.us-east-1.on.aws/}?file=${file.name}`);
            const { uploadURL } = await response.json();

            // 2. Upload directly to S3
            status.innerText = "Uploading to S3...";
            const upload = await fetch(uploadURL, {
                method: 'PUT',
                body: file
            });

            if (upload.ok) {
                status.innerText = "Success! Processing image... Check your Processed bucket in 5 seconds.";
            } else {
                status.innerText = "Upload failed.";
            }
        }
    </script>
