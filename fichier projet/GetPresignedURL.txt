import boto3
import json

def lambda_handler(event, context):
    s3 = boto3.client('s3', region_name='us-east-1')
    INPUT_BUCKET = 'project-label-image-receiver'
    PROCESSED_BUCKET = 'project-label-image-output'
    
    params = event.get('queryStringParameters', {})
    file_name = params.get('file')
    action = params.get('action', 'upload')

    try:
        if action == 'view':
            # ADD THE PREFIX HERE to match your AI Lambda's output
            processed_file_name = f"labeled-{file_name}"
            print(f"DEBUG: Looking for {processed_file_name} in {PROCESSED_BUCKET}")
            
            try:
                s3.head_object(Bucket=PROCESSED_BUCKET, Key=processed_file_name)
                url = s3.generate_presigned_url('get_object',
                    Params={'Bucket': PROCESSED_BUCKET, 'Key': processed_file_name},
                    ExpiresIn=60)
                status = "ready"
            except:
                status = "processing"
                url = None
        else:
            # Uploading uses the original name
            url = s3.generate_presigned_url('put_object',
                Params={'Bucket': INPUT_BUCKET, 'Key': file_name, 'ContentType': 'image/jpeg'},
                ExpiresIn=300)
            status = "upload_ready"

        return {
            "statusCode": 200,
            "headers": { "Access-Control-Allow-Origin": "*" },
            "body": json.dumps({"url": url, "status": status})
        }
    except Exception as e:
        return {"statusCode": 500, "body": json.dumps({"error": str(e)})}
