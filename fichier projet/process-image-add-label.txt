import boto3
from PIL import Image, ImageDraw
import io

# 1. Initialize clients with your region (e.g., 'us-east-1' or 'eu-west-1')
REGION = 'us-east-1' 
s3 = boto3.client('s3', region_name=REGION)
rekognition = boto3.client('rekognition', region_name=REGION)

def lambda_handler(event, context):
    # 2. Get bucket and file name from the S3 event
    # This comes from the 'Test' event you configured
    source_bucket = event['Records'][0]['s3']['bucket']['name']
    object_key = event['Records'][0]['s3']['object']['key']
    
    # 3. Download the image into memory
    print(f"Reading {object_key} from {source_bucket}")
    response = s3.get_object(Bucket=source_bucket, Key=object_key)
    img_bytes = response['Body'].read()
    
    # Open with Pillow to get dimensions
    img = Image.open(io.BytesIO(img_bytes))
    width, height = img.size

# 4. Call Rekognition using the S3 reference instead of raw bytes
    print(f"Asking Rekognition to look at: {source_bucket}/{object_key}")
    
    analysis = rekognition.detect_labels(
        Image={
            'S3Object': {
                'Bucket': source_bucket,
                'Name': object_key
            }
        },
        MaxLabels=10,
        MinConfidence=50
    )
    
    # Create a drawing handle
    draw = ImageDraw.Draw(img)

# ADD THIS LINE HERE:
    boxes_drawn = 0

# 5. Loop through labels
    boxes_drawn = 0
    for label in analysis['Labels']:
        label_name = label['Name']
        confidence = label['Confidence']
        
        if 'Instances' in label and label['Instances']:
            for instance in label.get('Instances', []):
                box = instance['BoundingBox']
                
                # Math for the box
                left = width * box['Left']
                top = height * box['Top']
                w = width * box['Width']
                h = height * box['Height']
                
                # 1. Draw the red rectangle
                points = (left, top, left + w, top + h)
                draw.rectangle(points, outline="red", width=5)
                
                # 2. Prepare the text (Name + %)
                display_text = f"{label_name} ({confidence:.1f}%)"
                
                # 3. Draw the text slightly above the box
                # We use (left, top - 20) so the text doesn't overlap the box line
                draw.text((left, top - 20), display_text, fill="red")
                
                boxes_drawn += 1
                print(f"Drew box and label for {label_name}")

    print(f"Total objects labeled: {boxes_drawn}")

    # 6. Save back to a buffer (RAM) and upload to the PROCESSED bucket
    buffer = io.BytesIO()
    img.save(buffer, format="JPEG")
    buffer.seek(0)
    
    # CHANGE THIS to your 3rd bucket name
    output_bucket = "project-label-image-output" 
    output_key = "labeled-" + object_key
    
    s3.put_object(Bucket=output_bucket, Key=output_key, Body=buffer, ContentType='image/jpeg')
    
    print(f"Success! Image saved to {output_bucket}/{output_key}")
    return {"status": "complete", "file": output_key}
