AWSTemplateFormatVersion: '2010-09-09'
Description: 'Infrastructure VisionCraft AI : Frontend, Upload et Processed'

Resources:
  # --- 1. BUCKET FRONTEND (Site Web) ---
  FrontendBucket:
    Type: AWS::S3::Bucket
    Properties:
      BucketName: !Sub 'visioncraft-frontend-${AWS::AccountId}'
      PublicAccessBlockConfiguration:
        BlockPublicAcls: false
        BlockPublicPolicy: false
        IgnorePublicAcls: false
        RestrictPublicBuckets: false
      WebsiteConfiguration:
        IndexDocument: index.html

  FrontendBucketPolicy:
    Type: AWS::S3::BucketPolicy
    Properties:
      Bucket: !Ref FrontendBucket
      PolicyDocument:
        Statement:
          - Effect: Allow
            Principal: "*"
            Action: s3:GetObject
            Resource: !Sub 'arn:aws:s3:::${FrontendBucket}/*'

  # --- 2. BUCKET UPLOAD (Images brutes) ---
  UploadBucket:
    Type: AWS::S3::Bucket
    Properties:
      BucketName: !Sub 'visioncraft-upload-${AWS::AccountId}'
      CorsConfiguration:
        CorsRules:
          - AllowedHeaders: ['*']
            AllowedMethods: [PUT, POST, GET]
            AllowedOrigins: ['*']

  # --- 3. BUCKET PROCESSED (Images avec labels) ---
  ProcessedBucket:
    Type: AWS::S3::Bucket
    Properties:
      BucketName: !Sub 'visioncraft-processed-${AWS::AccountId}'
      PublicAccessBlockConfiguration:
        BlockPublicAcls: false
        BlockPublicPolicy: false
      CorsConfiguration:
        CorsRules:
          - AllowedHeaders: ['*']
            AllowedMethods: [GET]
            AllowedOrigins: ['*']

  # --- 4. LAMBDA & PERMISSIONS ---
  LambdaExecutionRole:
    Type: AWS::IAM::Role
    Properties:
      AssumeRolePolicyDocument:
        Version: '2012-10-17'
        Statement:
          - Effect: Allow
            Principal:
              Service: [lambda.amazonaws.com]
            Action: ['sts:AssumeRole']
      Policies:
        - PolicyName: VisionCraftFullS3Access
          PolicyDocument:
            Version: '2012-10-17'
            Statement:
              - Effect: Allow
                Action: ['s3:PutObject', 's3:GetObject', 's3:ListBucket']
                Resource: 
                  - !Sub 'arn:aws:s3:::${UploadBucket}/*'
                  - !Sub 'arn:aws:s3:::${ProcessedBucket}/*'

  VisionCraftLambda:
    Type: AWS::Lambda::Function
    Properties:
      FunctionName: VisionCraft_Processor_V3
      Handler: index.lambda_handler
      Runtime: python3.9
      Role: !GetAtt LambdaExecutionRole.Arn
      Code:
        ZipFile: |
          def lambda_handler(event, context):
              return {"statusCode": 200, "body": "Ready"}

  LambdaUrl:
    Type: AWS::Lambda::Url
    Properties:
      TargetFunctionArn: !Ref VisionCraftLambda
      AuthType: NONE
      Cors:
        AllowOrigins: ['*']

  LambdaUrlPermission:
    Type: AWS::Lambda::Permission
    Properties:
      FunctionName: !Ref VisionCraftLambda
      Action: lambda:InvokeFunctionUrl
      Principal: '*'
      FunctionUrlAuthType: NONE

Outputs:
  WebsiteURL:
    Description: "URL de votre site web"
    Value: !GetAtt FrontendBucket.WebsiteURL
  LambdaAPI:
    Description: "URL à mettre dans le code JS"
    Value: !GetAtt LambdaUrl.FunctionUrl
